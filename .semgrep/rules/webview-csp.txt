function pages(csp: string) {
	// ruleid: insilode-webview-html-without-csp
	const bad = `<!doctype html><html><head><meta charset="utf-8"></head><body></body></html>`;
	// ok: insilode-webview-html-without-csp
	const good = `<!doctype html><html><head><meta charset="utf-8"><meta http-equiv="Content-Security-Policy" content="${csp}"></head></html>`;
	// ruleid: insilode-webview-html-without-csp
	const upper = `<!DOCTYPE html><html><head><meta charset="utf-8"></head><body></body></html>`;
	return [bad, good, upper];
}
